\section*{Overview}
This lecture covers:
\begin{itemize}
    \item Marginal PMF (definition, computation, example)
    \item Correlation and its limitations
    \item Covariance (definition, properties, interpretation)
    \item Pearson’s correlation coefficient
    \item Joint Gaussian random variables
\end{itemize}

\section*{Definitions and Notation}

\subsection*{Joint PMF}
For discrete random variables $X$ and $Y$:
\[
p_{XY}(x,y) = P(X=x, Y=y)
\]

\subsection*{Marginal PMFs}
\[
p_X(x) = \sum_y p_{XY}(x,y), \quad
p_Y(y) = \sum_x p_{XY}(x,y)
\]

\subsection*{Correlation (Raw Product Moment)}
\[
R_{XY} = E[XY]
\]

\subsection*{Covariance}
\[
\text{Cov}(X,Y) = E[(X - \mu_X)(Y - \mu_Y)]
\]

Equivalent form:
\[
\text{Cov}(X,Y) = E[XY] - E[X]E[Y]
\]

\[
\mu_X = E[X], \quad \mu_Y = E[Y]
\]

\subsection*{Pearson Correlation Coefficient}
\[
\rho_{XY} = \frac{\text{Cov}(X,Y)}{\sigma_X \sigma_Y}
\]

\[
-1 \leq \rho_{XY} \leq 1
\]

\section*{Joint Gaussian Distribution}
\[
f_{XY}(x,y) =
\frac{1}{2\pi \sigma_X \sigma_Y \sqrt{1-\rho^2}}
\exp\left(
-\frac{1}{2(1-\rho^2)}
\left[
\left(\frac{x-\mu_X}{\sigma_X}\right)^2
+
\left(\frac{y-\mu_Y}{\sigma_Y}\right)^2
-2\rho
\left(\frac{x-\mu_X}{\sigma_X}\right)
\left(\frac{y-\mu_Y}{\sigma_Y}\right)
\right]
\right)
\]

\section*{Marginal PMF}

\textbf{Key Idea:}  
Marginal PMF is obtained by summing over the other variable.

\[
p_X(x_i) = \sum_j p_{ij}, \quad
p_Y(y_j) = \sum_i p_{ij}
\]

\[
\sum_i p_X(x_i) = 1, \quad
\sum_j p_Y(y_j) = 1
\]

\section*{Correlation}

\begin{itemize}
    \item Measures $E[XY]$
    \item Depends on scale
    \item Computed about origin
    \item $R_{XY}=0$ implies orthogonality (not independence)
\end{itemize}

\section*{Covariance}

\subsection*{Interpretation}
\begin{itemize}
    \item Positive: same direction
    \item Negative: opposite direction
    \item Zero: no linear relation
\end{itemize}

\subsection*{Properties}
\begin{itemize}
    \item Symmetry:
    \[
    \text{Cov}(X,Y) = \text{Cov}(Y,X)
    \]

    \item Variance:
    \[
    \text{Cov}(X,X) = \text{Var}(X)
    \]

    \item Linearity:
    \[
    \text{Cov}(aX+b, cY+d) = ac \, \text{Cov}(X,Y)
    \]

    \item Independence:
    \[
    X \perp Y \Rightarrow \text{Cov}(X,Y) = 0
    \]
\end{itemize}

\textbf{Important:}
\[
\text{Cov}(X,Y)=0 \nRightarrow X \perp Y
\]

\section*{Pearson Coefficient}

\begin{itemize}
    \item Unit-free
    \item Measures strength of linear relationship
\end{itemize}

\[
\rho > 0 \Rightarrow \text{positive correlation}
\]
\[
\rho < 0 \Rightarrow \text{negative correlation}
\]
\[
\rho = 0 \Rightarrow \text{no linear correlation}
\]

\section*{Covariance Derivation}

\[
\text{Cov}(X,Y) = E[(X-\mu_X)(Y-\mu_Y)]
\]

\[
= E[XY - X\mu_Y - \mu_X Y + \mu_X \mu_Y]
\]

\[
= E[XY] - \mu_Y E[X] - \mu_X E[Y] + \mu_X \mu_Y
\]

\[
= E[XY] - \mu_X \mu_Y
\]

\[
= E[XY] - E[X]E[Y]
\]

\section*{Worked Example 1 (Marginal PMF)}

Joint PMF table:

\[
\begin{array}{c|cc}
 & y=0 & y=1 \\
\hline
x=0 & \frac{1}{8} & \frac{1}{4} \\
x=1 & \frac{3}{8} & \frac{1}{4}
\end{array}
\]

Marginals:

\[
p_X(0)=\frac{3}{8}, \quad p_X(1)=\frac{5}{8}
\]

\[
p_Y(0)=\frac{1}{2}, \quad p_Y(1)=\frac{1}{2}
\]

\section*{Worked Example 2 (Covariance)}

Given:
\[
Y = -2X + 3
\]

\[
\text{Cov}(X,Y) = -2 \, \text{Var}(X)
\]

\section*{Worked Example 3}

Given:
\[
f_{XY}(x,y) = \frac{xy}{96}, \quad 0<x<4, \; 1<y<5
\]

Results:
\[
E[X] = \frac{8}{3}, \quad E[Y] = \frac{31}{9}
\]

\[
\text{Var}(X) = \frac{8}{9}, \quad \text{Var}(Y) = \frac{92}{81}
\]

\[
\text{Cov}(X,Y) = 0
\]

\[
\rho_{XY} = 0
\]

\section*{Summary}

\begin{itemize}
    \item Marginal PMF:
    \[
    p_X(x)=\sum_y p_{XY}(x,y), \quad
    p_Y(y)=\sum_x p_{XY}(x,y)
    \]

    \item Correlation:
    \[
    R_{XY}=E[XY]
    \]

    \item Covariance:
    \[
    \text{Cov}(X,Y)=E[XY]-E[X]E[Y]
    \]

    \item Pearson coefficient:
    \[
    \rho_{XY}=\frac{\text{Cov}(X,Y)}{\sigma_X \sigma_Y}
    \]

    \item Key Result:
    \[
    \text{Cov}(X,Y)=0 \nRightarrow X \perp Y
    \]
\end{itemize}
